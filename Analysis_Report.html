# Overall summary
```{r}
#| label: n-metaclusters

cleaned_data <- read_csv(csv_files[str_detect(csv_files, "all")])
n_metaclusters <- n_distinct(cleaned_data$FlowSOM_metacluster, na.rm = TRUE)
n_samples <- n_distinct(cleaned_data$Sample, na.rm = TRUE)
```

```{r}
#| label: summary-table

summary_table <- read_csv(csv_files[str_detect(csv_files, "Summary")])

# Convert the per-cluster percentage columns from wide to long format
cluster_percentages <- summary_table |>
  pivot_longer(
    cols = matches("^Percent of sample -- \\d+$"),
    names_to = "Cluster",
    names_pattern = "^Percent of sample -- (\\d+)$",
    values_to = "Percent"
  ) |>
  mutate(
    Cluster = paste("Metacluster", Cluster),
    Percent = as.numeric(Percent)
  )

# Keep the top clusters for each sample
top_clusters_number <- analysis_parameters$top_number_metaclusters_rep

top_clusters <- cluster_percentages |>
  group_by(Sample) |>
  slice_max(order_by = Percent, n = top_clusters_number, with_ties = FALSE) |>
  arrange(desc(Percent), .by_group = TRUE) |>
  summarise(
    `Top represented clusters` = paste(
      sprintf("%s (%.1f%%)", Cluster, Percent),
      collapse = "; "
    ),
    .groups = "drop"
  )

# Count cells in the cleaned population, then make the overview table
overview_table <- cleaned_data |>
  count(Sample, name = "Cells per sample, cleaned") |>
  left_join(top_clusters, by = "Sample")

```
