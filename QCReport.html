function qcCarouselMove(carouselId, direction) {
  const carousel = document.getElementById(carouselId);

  if (!carousel) {
    return;
  }

  const track = carousel.querySelector(".qc-carousel-track");
  const slides = carousel.querySelectorAll(".qc-carousel-slide");
  const counter = carousel.querySelector(".qc-carousel-counter");

  if (slides.length === 0) {
    return;
  }

  let currentIndex = Number(carousel.dataset.currentIndex || 0);

  currentIndex = currentIndex + direction;

  if (currentIndex < 0) {
    currentIndex = slides.length - 1;
  }

  if (currentIndex >= slides.length) {
    currentIndex = 0;
  }

  carousel.dataset.currentIndex = currentIndex;
  track.style.transform = "translateX(-" + (currentIndex * 100) + "%)";
  counter.textContent = (currentIndex + 1) + " / " + slides.length;
}

find_images <- function(path) {
  if (!dir.exists(path)) {
    return(character())
  }

  sort(
    list.files(
      path = path,
      pattern = "\\.(jpg|jpeg|png)$",
      full.names = TRUE,
      ignore.case = TRUE
    )
  )

## Helper for tibble reading
summary_value <- function(data, column, default = "Unavailable") {
  if (
    is.null(data) ||
      nrow(data) == 0 ||
      !column %in% names(data) ||
      is.na(data[[column]][[1]])
  ) {
    return(default)
  }

  as.character(data[[column]][[1]])
}

# Status-summary helpers

count_status <- function(data, status) {
  if (
    is.null(data) ||
      nrow(data) == 0 ||
      !"status" %in% names(data)
  ) {
    return(0L)
  }

  sum(
    data$status == status,
    na.rm = TRUE
  )
}

has_compensaid_flags <- function(data) {
  if (
    is.null(data) ||
      nrow(data) == 0 ||
      !"flagged_pairs" %in% names(data)
  ) {
    return(FALSE)
  }

  flagged_pairs <- suppressWarnings(
    as.numeric(data$flagged_pairs)
  )

  any(
    flagged_pairs > 0,
    na.rm = TRUE
  )
}


# HTML carousel helper

make_image_carousel <- function(
  image_files,
  carousel_id,
  title = NULL,
  image_height = "32rem"
) {
  image_files <- image_files[
    file.exists(image_files)
  ]

  if (length(image_files) == 0) {
    return(
      htmltools::HTML(
        "<p>No images were found for this section.</p>"
      )
    )
  }

  image_files <- normalizePath(
    image_files,
    winslash = "/",
    mustWork = TRUE
  )

  image_sources <- fs::path_rel(
    image_files,
    start = project_root
  )

  image_sources <- gsub(
    "\\\\",
    "/",
    image_sources
  )

  escape_html <- function(x) {
    x <- gsub("&", "&amp;", x, fixed = TRUE)
    x <- gsub("<", "&lt;", x, fixed = TRUE)
    x <- gsub(">", "&gt;", x, fixed = TRUE)
    x <- gsub("\"", "&quot;", x, fixed = TRUE)
    x
  }

  slides <- purrr::map2_chr(
    image_sources,
    basename(image_files),
    function(image_source, image_label) {
      paste0(
        '<div class="qc-carousel-slide">',
        '<img src="',
        escape_html(image_source),
        '" alt="',
        escape_html(image_label),
        '" style="height:',
        image_height,
        ';">',
        '<div class="qc-carousel-caption">',
        escape_html(image_label),
        "</div>",
        "</div>"
      )
    }
  )

  title_html <- if (is.null(title)) {
    ""
  } else {
    paste0(
      "<h4>",
      escape_html(title),
      "</h4>"
    )
  }

  html <- paste0(
    '<div class="qc-carousel" id="',
    escape_html(carousel_id),
    '">',

    title_html,

    '<button class="qc-carousel-button previous" ',
    'type="button" aria-label="Previous image" ',
    'onclick="qcCarouselMove(\'',
    escape_html(carousel_id),
    '\', -1)">',
    "&#10094;",
    "</button>",

    '<div class="qc-carousel-window">',
    '<div class="qc-carousel-track">',
    paste(slides, collapse = ""),
    "</div>",
    "</div>",

    '<button class="qc-carousel-button next" ',
    'type="button" aria-label="Next image" ',
    'onclick="qcCarouselMove(\'',
    escape_html(carousel_id),
    '\', 1)">',
    "&#10095;",
    "</button>",

    '<div class="qc-carousel-counter">',
    "1 / ",
    length(slides),
    "</div>",

    "</div>"
  )

  htmltools::HTML(html)
}

af_has_flags <- function(data) {
  if (
    is.null(data) ||
      nrow(data) == 0 ||
      !"af_status" %in% names(data)
  ) {
    return(FALSE)
  }

  any(
    data$af_status == "flagged",
    na.rm = TRUE
  )
