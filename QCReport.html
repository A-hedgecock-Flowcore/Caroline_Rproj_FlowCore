function qcCarouselMove(carouselId, direction) {
  const carousel = document.getElementById(carouselId);

  if (!carousel) {
    return;
  }

  const track = carousel.querySelector(".qc-carousel-track");
  const slides = carousel.querySelectorAll(".qc-carousel-slide");
  const counter = carousel.querySelector(".qc-carousel-counter");

  if (slides.length === 0) {
    return;
  }

  let currentIndex = Number(carousel.dataset.currentIndex || 0);

  currentIndex = currentIndex + direction;

  if (currentIndex < 0) {
    currentIndex = slides.length - 1;
  }

  if (currentIndex >= slides.length) {
    currentIndex = 0;
  }

  carousel.dataset.currentIndex = currentIndex;
  track.style.transform = "translateX(-" + (currentIndex * 100) + "%)";
  counter.textContent = (currentIndex + 1) + " / " + slides.length;
}

status_badge <- function(status) {
  if (identical(status, "completed")) {
    return("✅ Completed")
  }

  if (identical(status, "failed")) {
    return("❌ Failed")
  }

  if (identical(status, "flagged")) {
    return("⚠️ Flagged")
  }

  "Unknown"
}

count_status <- function(data, status) {
  if (is.null(data) || nrow(data) == 0) {
    return(0L)
  }

  sum(
    data$status == status,
    na.rm = TRUE
  )
}

compensaid_summary <- function(results) {
  if (is.null(results) || length(results) == 0) {
    return(
      tibble::tibble(
        sample = character(),
        flagged_pairs = integer(),
        review_status = character()
      )
    )
  }

  purrr::imap_dfr(
    results,
    function(result, sample_name) {
      flagged_pairs <- result$flagged_pairs

      n_flagged <- if (is.null(flagged_pairs)) {
        0L
      } else {
        nrow(flagged_pairs)
      }

      tibble::tibble(
        sample = sample_name,
        flagged_pairs = n_flagged,
        review_status = if (n_flagged > 0) {
          "Review"
        } else {
          "No flagged pairs"
        }
      )
    }
  )
}

patient_compensaid_summary <- compensaid_summary(
  patient_compensaid_results
)

  controls_compensaid_results

make_image_carousel <- function(
  image_files,
  carousel_id,
  title = NULL,
  image_height = "32rem"
) {
  image_files <- image_files[
    file.exists(image_files)
  ]

  if (length(image_files) == 0) {
    return(
      htmltools::HTML(
        "<p>No images were found for this section.</p>"
      )
    )
  }

  image_files <- normalizePath(
    image_files,
    winslash = "/",
    mustWork = TRUE
  )

  image_sources <- fs::path_rel(
    image_files,
    start = project_root
  )

  image_sources <- gsub(
    "\\\\",
    "/",
    image_sources
  )

  escape_html <- function(x) {
    x <- gsub("&", "&amp;", x, fixed = TRUE)
    x <- gsub("<", "&lt;", x, fixed = TRUE)
    x <- gsub(">", "&gt;", x, fixed = TRUE)
    x <- gsub("\"", "&quot;", x, fixed = TRUE)
    x
  }

  slides <- purrr::map2_chr(
    image_sources,
    basename(image_files),
    function(image_source, image_label) {
      paste0(
        '<div class="qc-carousel-slide">',
        '<img src="',
        escape_html(image_source),
        '" alt="',
        escape_html(image_label),
        '" style="height:',
        image_height,
        ';">',
        '<div class="qc-carousel-caption">',
        escape_html(image_label),
        "</div>",
        "</div>"
      )
    }
  )

  title_html <- if (is.null(title)) {
    ""
  } else {
    paste0(
      "<h4>",
      escape_html(title),
      "</h4>"
    )
  }

  html <- paste0(
    '<div class="qc-carousel" id="',
    escape_html(carousel_id),
    '">',

if (is.null(patient_compensaid_results)) {

# Review checklist

Use this checklist during review:

- [ ] Pipeline completed without a fatal error.
- [ ] All expected patient samples are present in the input inventory.
- [ ] Patient unstained samples are available for the intended patient samples.
- [ ] AF extraction flags have been reviewed.
- [ ] Patient unmixing flags and failures have been reviewed.
- [ ] Nonpatient unmixing failures have been reviewed.
- [ ] Patient cleanup gates appear appropriate.
- [ ] Control cleanup gates appear appropriate.
- [ ] Event populations are plausible after cleanup gating.
- [ ] compensAID flagged pairs have been reviewed.
