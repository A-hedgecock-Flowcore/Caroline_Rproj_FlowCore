#| label: setup

library(tidyverse)

project_root <- getwd()

analysisInfo_folder <- file.path("AnalysisInfo")
parameters_file <- list.files(
  analysisInfo_folder,
  "\\.yml$",
  full.names = TRUE,
  ignore.case = TRUE
)
analysis_parameters <- yaml::read_yaml(parameters_file)
output_folder <- file.path(project_root, analysis_parameters$output_folder)
spectre_output_folder <- file.path(output_folder, "Spectre_outputs")

if (!fs::is_absolute_path(output_folder)) {
  output_folder <- file.path(
    project_root,
    output_folder
  )
}

if (!dir.exists(project_root, output_folder)) {
  stop(
    "Output folder does not exist: ",
    output_folder,
    " \n QCPipeline.R and AnalysisPipeline.R may not have run yet."
  )
}

# output files
analysis_outputs <- list.files(spectre_output_folder, full.names = TRUE)

csv_files <- analysis_outputs[str_detect(analysis_outputs, "\\.csv$")]
image_files <- analysis_outputs[str_detect(analysis_outputs, "\\.png$")]
